Public Sub METEO()
'Macro initieren met naam
'=================================================================
    Application.StatusBar = "Voorbereiden"
    Application.EnableEvents = False
    MCommon.Macro_Start "Meteo"
    Sheets(D_Name).Select

'Berekeningen en opmaak kolommen
'=================================================================
    Application.StatusBar = "Berekeningen Maken"
'   Kolom DV3 naast DV1 en DV2 en LFW3 verwijderen
    MKolom.Verwijderen "AV AS"
    MKolom.Verplaatsen "AU", "AO", D_Name

'   Tijd_halfuur omzetting voor zonnestraling
    MKolom.Formule "AL", "=TIME(0,TRUNC(AL1*30),ROUND((AL1*30-TRUNC(AL1*30))*60,0))", "mm:ss", , 1, -1
    
'   Windrichting berekenen
    Graden = "AG"
    MKolom.Windrichting Graden
   
'   Omzetten van de bladnatheid naar percentage.
    MKolom.Numberformat "W AK AL AU AV", "0%"
    
'   Omzetten van de Datum.
    MKolom.Numberformat "AJ", "dd/mm/yy hh:mm"
        
'   Bepalen zon op of onder op basis van Zonnestraling > 5 W/m²
    MKolom.Formule "AW", "=IF(P1>5,30,0)", "mm:ss"
    
'   verschil tussen kort in (CNR1) en zon (CMP6)
    MKolom.Formule "AX", "=P1 - Z1", "0.00"
    
'   Albedo onderaan afknippen als Kort In < 5W
    MKolom.Formule "W", "=IF(P1 < 5,0,W1)", "0%", , 1, -1

'   Conditionele opmaak Bladnat sensoren
    MKolom.CndOpmaakoud "AU:AV,AK:AL"
    
'   Kolombreedte op 10 AJ op 15 zetten
    'ActiveSheet.Range("AJ:AJ").ColumnWidth = 18
    
'   Berekening "Leaf T" in Laatste Kolom
    MKolom.Formule "AY", "=(S1/0.0000000567)^(1/4)-273.16", "0.000"
   
    
'Titels plaatsen en eventueel Maskeren
'================================================================
    Application.StatusBar = "Titels Filteren en Plaatsen"
    MCommon.Maskeren
    'Titels Bijkomende berekeningen
    With Worksheets(D_Name)
        .Range("AY1").Value = "Leaf T."
        .Range("AM1").Value = "Sun Half Hours"
        .Range("AW1").Value = "DayLight"
        .Range("AX1").Value = "Solar_R_I - SOLAR_RAD"
    End With
    MCommon.Titels_ZoekEnPlaats
'GRAFIEKEN
'================================================================
    Application.StatusBar = "Grafieken Maken"

    'Wind gerelateerde grafieken
    Tabkleur = 8    'Turkoise
    MGraphs.Builder "Wind", DATAsht.Name, xlLine, "A AC AD AF ai", "", "", "", "Gesplitst"
    MGraphs.Builder "Wind 40 m", DATAsht.Name, xlLine, "A AE AF", "", "", "", "False"
    MGraphs.Builder "VGL Wind 40m", DATAsht.Name, xlXYScatter, "AF AE", "", "", "", "False"
    MGraphs.Builder "WindRichting", DATAsht.Name, xlLine, "A AG", "360", "0", "", "False"
    MGraphs.Builder "Luchtdruk", DATAsht.Name, xlLine, "A O", "", "", "", "False"
    
    'Temperatuur gerelateerde grafieken
    Tabkleur = 3    'Rood
    MGraphs.Builder "Droge T", DATAsht.Name, xlLine, "A G H I ay", "", "", "", "Gesplitst"
    MGraphs.Builder "Bodem T", DATAsht.Name, xlLine, "A C D x y aa", "", "", "", "Gesplitst"
    MGraphs.Builder "Cabine T", DATAsht.Name, xlLine, "A F", "", "", "", "False"
    
    'Neerslag gerelateerde grafieken
    Tabkleur = 5    'Blauw
    MGraphs.Builder "Neerslag", DATAsht.Name, xlLine, "A aq AN AO AP", "", "", "", "Gesplitst"
    MGraphs.Builder "Decagon", DATAsht.Name, xlLine, "A aq AS AT", "", "", "", "Gesplitst"
    MGraphs.Builder "Bladnat", DATAsht.Name, xlLine, "A aq AK AL", "", "", "", "Gesplitst"
    MGraphs.Builder "RV", DATAsht.Name, xlLine, "A J K L", "", "", "", "False"

    Tabkleur = 6    'Geel
    MGraphs.Builder "Straling", DATAsht.Name, xlLine, "A Z P aa v", "", "", "", "False"
    MGraphs.Builder "Albedo", DATAsht.Name, xlLine, "A W", "", "", "", "False"

Public Sub WX_Station(plot)

'Macro initieren met naam
'=================================================================
    Application.EnableEvents = False
    Application.StatusBar = "Voorbereiden"
    MCommon.Macro_Start "WX_" & plot

    Sheets(D_Name).Select
    Select Case plot
        Case "ZON", "WIJ"
            MKolom.Invoegen "O"
            'MKolom.Verwijderen "C"  'PlotNummer vewijderen
        Case "GRIM"
            MKolom.Verwijderen "O N L K"
            MKolom.Invoegen "P P P P P"
        Case "BRA"
            'MKolom.Verwijderen "O M L"
            'MKolom.Invoegen "M M M M M C"
            If [C1] <> "Plot" Then MKolom.Invoegen "C"
    End Select
    

'Berekeningen en opmaak kolommen
'=================================================================
    Application.StatusBar = "Berekeningen Maken"
'Windrichting berekenen
 'Eenheid = "F"

'================================================================

    MCommon.Maskeren

    MGraphs.Builder "RV", DATAsht.Name, xlLine, "A J", "", "", "", "False"

    MGraphs.Builder "Neerslag", DATAsht.Name, xlLine, "A L q r", "", "", "", "Zelfde Nul-As"

    Case "WIJ", "ZON", "GRIM"
    'Naargelang Straling voorhanden is

    MGraphs.Builder "Uren Zon", DATAsht.Name, xlColumnClustered, "A X", "", "", "", "False"
